#if UNITY_EDITOR
namespace TiltBrush.Benchmarks
{
    public static class BatchWorkloadCounters
    {
        public static readonly string[] Names = { "Batch.FullUpload", "Batch.TopologyUpload", "GeometryBrush.VisualUpdate", "GeometryPool.FullCopy", "GeometryPool.FullReload", "GeometryPool.IndexRead", "GeometryPool.TriangleIds", "Pointer.ControlPoint", "Pointer.Recreate", "GeometryPool.SubsetCopy", "Batch.Append", "Pointer.VisualUpdate", "Pointer.BulkControlPoints", "Pointer.FinalPreparation" };
        public static bool Enabled;
        public static readonly long[] Calls = new long[14], Ticks = new long[14], Vertices = new long[14], Indices = new long[14];
        public static void Reset()
        {
            System.Array.Clear(Calls, 0, Calls.Length); System.Array.Clear(Ticks, 0, Ticks.Length);
            System.Array.Clear(Vertices, 0, Vertices.Length); System.Array.Clear(Indices, 0, Indices.Length);
        }
        public static Scope Measure(int id, int vertices = 0, int indices = 0) => new Scope(id, vertices, indices);
        public struct Scope : System.IDisposable
        {
            readonly int m_Id;
            readonly long m_Start;
            public Scope(int id, int vertices, int indices)
            {
                m_Id = Enabled ? id : -1;
                m_Start = m_Id < 0 ? 0 : System.Diagnostics.Stopwatch.GetTimestamp();
                if (m_Id >= 0) { Calls[id]++; Vertices[id] += vertices; Indices[id] += indices; }
            }
            public void Dispose()
            {
                if (m_Id >= 0) Ticks[m_Id] += System.Diagnostics.Stopwatch.GetTimestamp() - m_Start;
            }
        }
    }
}
#endif
